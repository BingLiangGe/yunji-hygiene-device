编辑
ServerBootstrap b = new ServerBootstrap();
b.group(bossGroup, workerGroup)
 .channel(NioServerSocketChannel.class)
 .option(ChannelOption.SO_BACKLOG, 1024) // 控制 Listen backlog 队列（全连接队列最大长度）
 .option(ChannelOption.SO_REUSEADDR, true) // 快速重启时端口复用
 .childOption(ChannelOption.TCP_NODELAY, true) // 关闭 Nagle 算法，减少延迟
 .childOption(ChannelOption.SO_KEEPALIVE, true) // 启用 TCP keepalive 探测
 .childOption(ChannelOption.SO_RCVBUF, 1024 * 128) // 接收缓冲区
 .childOption(ChannelOption.SO_SNDBUF, 1024 * 128) // 发送缓冲区
 .childOption(ChannelOption.WRITE_BUFFER_WATER_MARK,
     new WriteBufferWaterMark(32 * 1024, 64 * 1024)) // 写缓冲水位线，防止内存暴涨
 .childHandler(new ChannelInitializer<SocketChannel>() {
     @Override
     protected void initChannel(SocketChannel ch) {
         ch.pipeline()
             .addLast(new MyBusinessHandler()); // 自定义处理器
     }
 });
✅ 二、Linux 系统参数优化建议（/etc/sysctl.conf）
bash
复制
编辑
# 允许更多连接排队
net.core.somaxconn = 1024
net.ipv4.tcp_max_syn_backlog = 8192

# 启用并配置 TCP keepalive
net.ipv4.tcp_keepalive_time = 60
net.ipv4.tcp_keepalive_intvl = 10
net.ipv4.tcp_keepalive_probes = 3

# 提高缓冲区
net.core.rmem_default = 262144
net.core.rmem_max = 4194304
net.core.wmem_default = 262144
net.core.wmem_max = 4194304

# 允许更多端口临时重用（避免 TIME_WAIT 累积）
net.ipv4.tcp_tw_reuse = 1
net.ipv4.tcp_fin_timeout = 15
